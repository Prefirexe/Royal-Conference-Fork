military_artillery_cannon_generic_diorama_entity = {
	entity = "military_artillery_cannon_generic_entity"

	group = "artillery"
	
	trigger = {
		scope:military_formation = {
			has_mobilization_option = mobilization_option:UHW_mo_cannon_artillery
			NOR = {
				has_mobilization_option = mobilization_option:UHW_mo_mobile_artillery
				has_mobilization_option = mobilization_option:UHW_mo_shrapnel_artillery
				has_mobilization_option = mobilization_option:UHW_mo_rapid_fire_artillery
				has_mobilization_option = mobilization_option:UHW_mo_siege_artillery
				has_mobilization_option = mobilization_option:UHW_mo_advanced_artillery
			}
		}
	}
}
